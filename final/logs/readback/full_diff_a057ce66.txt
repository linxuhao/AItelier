+"""Run ONE command and write its log file; this is how every log under
+`final/logs/` except `final/logs/mutations/` was produced (the mutation
+runner writes its own).
+
+    python3 final/capture.py <log> [--container NAME] [--cwd DIR] \
+        [ENV=VAL ...] -- <cmd ...>
+
+The log holds: the command, the real cwd, the tree's HEAD and porcelain line
+count, the runner (the full `docker run` argv, or `host`), the environment
+overrides, the sha256 of every argument that is an existing file (probe
+scripts), start and end time, the command's combined output, and
+`BARE_RC=<exit code>`, the exit code of the command itself. capture.py exits
+with that code.
+
+With `--container NAME` the command runs in a throwaway container
+(`docker run --rm --init -m 3g`, never the production `aitelier`
+container), with the repository and `~/.AItelier` mounted at their host
+paths so a git worktree's gitdir resolves. Without it, the command runs on
+the host.
+"""
+from __future__ import annotations
+
+import datetime
+import hashlib
+import os
+import subprocess
+import sys
+from pathlib import Path
+
+HOME = "/home/linxuhao"
+IMAGE = "aitelier:latest"
+
+
+def _now() -> str:
+    return datetime.datetime.now(datetime.timezone.utc).strftime(
+        "%Y-%m-%dT%H:%M:%SZ")
+
+
+def _git(cwd: str, *args: str) -> str:
+    proc = subprocess.run(["git", "-C", cwd, *args], capture_output=True,
+                          text=True)
+    return proc.stdout.strip() if proc.returncode == 0 else f"(git rc={proc.returncode})"
+
+
+def main(argv: list[str]) -> int:
+    log = Path(argv[0]).resolve()
+    argv = argv[1:]
+    container = None
+    cwd = os.getcwd()
+    env: dict[str, str] = {}
+    while argv and argv[0] != "--":
+        head = argv.pop(0)
+        if head == "--container":
+            container = argv.pop(0)
+        elif head == "--cwd":
+            cwd = str(Path(argv.pop(0)).resolve())
+        else:
+            key, _, value = head.partition("=")
+            env[key] = value
+    cmd = argv[1:]
+    if container:
+        base_env = {"HOME": HOME, "PYTHONPATH": cwd,
+                    "PYTHONDONTWRITEBYTECODE": "1",
+                    "SEARXNG_URL": "http://linxuhaserver:8888"}
+        base_env.update(env)
+        run = ["docker", "run", "--rm", "--init", "-m", "3g",
+               "--name", container, "--network", "aitelier_default",
+               "-u", "1000:1000"]
+        for key, value in base_env.items():
+            run += ["-e", f"{key}={value}"]
+        run += ["-v", f"{HOME}/AItelier:/app",
+                "-v", f"{HOME}/AItelier:{HOME}/AItelier",
+                "-v", f"{HOME}/.AItelier:{HOME}/.AItelier",
+                "-w", cwd, IMAGE, *cmd]
+        runner = " ".join(run)
+        child_env = None
+    else:
+        run = cmd
+        runner = "host"
+        child_env = {**os.environ, **env}
+    hashes = []
+    for arg in cmd:
+        path = Path(cwd, arg) if not os.path.isabs(arg) else Path(arg)
+        if path.is_file():
+            digest = hashlib.sha256(path.read_bytes()).hexdigest()
+            hashes.append(f"{arg} sha256={digest}")
+    started = _now()
+    proc = subprocess.run(run, cwd=cwd, env=child_env, capture_output=True,
+                          text=True)
+    ended = _now()
+    porcelain = _git(cwd, "status", "--porcelain")
+    header = [
+        f"# CMD: {' '.join(cmd)}",
+        f"# CWD: {cwd}",
+        f"# HEAD: {_git(cwd, 'rev-parse', 'HEAD')} "
+        f"porcelain_lines={len(porcelain.splitlines()) if porcelain else 0}",
+        f"# RUNNER: {runner}",
+        f"# ENV: {' '.join(f'{k}={v}' for k, v in env.items()) or '(none)'}",
+        *(f"# FILE: {h}" for h in hashes),
+        f"# STARTED: {started}",
+    ]
+    log.parent.mkdir(parents=True, exist_ok=True)
+    log.write_text("\n".join(header) + "\n" + proc.stdout + proc.stderr
+                   + f"# ENDED: {ended}\nBARE_RC={proc.returncode}\n",
+                   encoding="utf-8")
+    print(f"{log.name} BARE_RC={proc.returncode}")
+    return proc.returncode
+
+
+if __name__ == "__main__":
+    raise SystemExit(main(sys.argv[1:]))

+"""pytest plugin loaded by `run_mutations.py` as ``-p ignition_plugin``.
+
+It measures two things about ONE applied mutation, inside the pytest process:
+
+* ``line_hits`` — how many times the mutated lines executed (Python's
+  ``sys.monitoring`` LINE events, restricted to the replacement's lines; every
+  other code location is disabled after its first event, so the cost is one
+  callback per location), and ``igniting_tests``, the tests during whose
+  call phase a mutated line executed;
+* ``source_readers`` — which tests read a mutated file's TEXT while they ran
+  (``open`` / ``io.open`` / ``tokenize.open`` / ``linecache.getlines`` during
+  the test's call phase). A test that went red after reading the mutated
+  source is a source-text witness, not a behavioural one; ``file_reads``
+  counts those reads. A test in both lists executed the mutated code, so it
+  is not a source-text witness.
+
+Input: ``MUTATION_IGNITION_SPEC`` = JSON ``{"lines": {path: [line, ...]},
+"files": [path, ...]}``. Output: ``MUTATION_IGNITION_OUT`` = JSON, written at
+session end. Code run in a child process is not counted.
+"""
+from __future__ import annotations
+
+import builtins
+import io
+import json
+import linecache
+import os
+import sys
+import tokenize
+
+import pytest
+
+_SPEC = json.loads(os.environ.get("MUTATION_IGNITION_SPEC") or "{}")
+_OUT = os.environ.get("MUTATION_IGNITION_OUT") or ""
+_LINES = {os.path.realpath(p): frozenset(v)
+          for p, v in (_SPEC.get("lines") or {}).items()}
+_FILES = frozenset(os.path.realpath(p) for p in (_SPEC.get("files") or []))
+
+_hits = {path: 0 for path in _LINES}
+_reads = {path: 0 for path in _FILES}
+_readers: set[str] = set()
+_igniters: set[str] = set()
+_current: list[str] = []
+_resolved: dict[str, str | None] = {}
+
+
+def _target_of(filename) -> str | None:
+    key = str(filename)
+    if key not in _resolved:
+        try:
+            real = os.path.realpath(os.fspath(filename))
+        except (TypeError, ValueError):
+            real = None
+        _resolved[key] = real
+    return _resolved[key]
+
+
+_mon = sys.monitoring
+_TOOL = next((t for t in (4, 3) if _mon.get_tool(t) is None), None)
+
+
+def _on_line(code, line):
+    path = _target_of(code.co_filename)
+    lines = _LINES.get(path)
+    if lines is None or line not in lines:
+        return _mon.DISABLE
+    _hits[path] += 1
+    if _current:
+        _igniters.add(_current[-1])
+    return None
+
+
+if _TOOL is not None and _LINES:
+    _mon.use_tool_id(_TOOL, "mutation-ignition")
+    _mon.register_callback(_TOOL, _mon.events.LINE, _on_line)
+    _mon.set_events(_TOOL, _mon.events.LINE)
+
+
+def _note_read(filename) -> None:
+    if not _current:
+        return
+    path = _target_of(filename)
+    if path in _reads:
+        _reads[path] += 1
+        _readers.add(_current[-1])
+
+
+def _wrap_open(real):
+    def opener(file, *args, **kwargs):
+        if isinstance(file, (str, bytes, os.PathLike)):
+            _note_read(file)
+        return real(file, *args, **kwargs)
+    return opener
+
+
+def _wrap_getlines(real):
+    def getlines(filename, module_globals=None):
+        _note_read(filename)
+        return real(filename, module_globals)
+    return getlines
+
+
+@pytest.hookimpl(hookwrapper=True)
+def pytest_runtest_call(item):
+    saved = (builtins.open, io.open, tokenize.open, linecache.getlines)
+    builtins.open = io.open = _wrap_open(saved[0])
+    tokenize.open = _wrap_open(saved[2])
+    linecache.getlines = _wrap_getlines(saved[3])

+    try:

+    finally:
+        _current.pop()
+        builtins.open, io.open, tokenize.open, linecache.getlines = saved
+
+

+            "mutated_modules_imported": imported,
